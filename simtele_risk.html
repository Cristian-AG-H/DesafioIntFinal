SCRIPT DDL SQL POSTGRESQL / SQLITE:
CREATE TABLE caso_estudio (
    id VARCHAR(32) PRIMARY KEY,
    nombre VARCHAR(120) NOT NULL,
    modelo_version VARCHAR(20) NOT NULL
);

CREATE TABLE escenario (
    id VARCHAR(48) PRIMARY KEY,
    caso_id VARCHAR(32) REFERENCES caso_estudio(id),
    tipo VARCHAR(20) CHECK (tipo IN ('NORMAL', 'ESTRES', 'MAL_CONDICIONADO')),
    parametros_json TEXT NOT NULL,
    fuente_datos VARCHAR(255)
);

CREATE TABLE simulacion (
    id VARCHAR(64) PRIMARY KEY,
    escenario_id VARCHAR(48) REFERENCES escenario(id),
    fecha TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    usuario VARCHAR(64),
    estado VARCHAR(20) CHECK (estado IN ('EXITOSA', 'FALLIDA'))
);

CREATE TABLE ejecucion_metodo (
    id VARCHAR(64) PRIMARY KEY,
    simulacion_id VARCHAR(64) REFERENCES simulacion(id),
    metodo VARCHAR(64) NOT NULL,
    tolerancia DOUBLE PRECISION,
    paso DOUBLE PRECISION,
    max_iter INTEGER,
    convergio BOOLEAN,
    tiempo_ms DOUBLE PRECISION
);

CREATE TABLE iteracion_resultado (
    id SERIAL PRIMARY KEY,
    ejecucion_id VARCHAR(64) REFERENCES ejecucion_metodo(id),
    numero INTEGER NOT NULL,
    aproximacion_json TEXT,
    residual DOUBLE PRECISION,
    error_abs DOUBLE PRECISION,
    error_rel DOUBLE PRECISION
);

CREATE TABLE resultado_resumen (
    id VARCHAR(64) PRIMARY KEY,
    ejecucion_id VARCHAR(64) UNIQUE REFERENCES ejecucion_metodo(id),
    referencia DOUBLE PRECISION,
    valor_json TEXT,
    metrica_error_json TEXT,
    interpretacion TEXT
);
